{{- else }}

Get the application URL by running these commands:

  # Web UI
  kubectl port-forward svc/{{ include "postman-clone.web.fullname" . }} 8080:{{ .Values.web.service.port }} -n {{ include "postman-clone.namespace" . }}
  # Then open http://localhost:8080

  # API (optional)
  kubectl port-forward svc/{{ include "postman-clone.api.fullname" . }} 4000:{{ .Values.api.service.port }} -n {{ include "postman-clone.namespace" . }}
  # Then open http://localhost:4000/api

{{- end }}
